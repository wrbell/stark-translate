"""Test-only in-memory restoration of the former queue wait; never edits source."""

import ast
from pathlib import Path

import pytest


@pytest.fixture(autouse=True)
def restore_legacy_queue_wait(monkeypatch):
    import dry_run_ab as pipeline

    path = Path(pipeline.__file__)
    tree = ast.parse(path.read_text())
    function = next(node for node in tree.body if isinstance(node, ast.AsyncFunctionDef) and node.name == "audio_loop")

    class RestoreWaitFor(ast.NodeTransformer):
        replaced = 0

        def visit_AsyncWith(self, node):
            if ast.unparse(node.items[0].context_expr) == "asyncio.timeout(0.1)":
                self.replaced += 1
                return ast.parse("audio_frame = await asyncio.wait_for(audio_queue.get(), timeout=0.1)").body[0]
            return self.generic_visit(node)

    transform = RestoreWaitFor()
    modified = transform.visit(function)
    assert transform.replaced == 1
    module = ast.fix_missing_locations(ast.Module(body=[modified], type_ignores=[]))
    original = pipeline.audio_loop
    exec(compile(module, str(path), "exec"), vars(pipeline))
    replacement = pipeline.audio_loop
    pipeline.audio_loop = original
    monkeypatch.setattr(pipeline, "audio_loop", replacement)
