bindings = {}


def read(path):
    data = path.read_bytes()
    bindings[str(path)] = {"sha256": hashlib.sha256(data).hexdigest(), "size_bytes": len(data)}
    return json.loads(data)


def source(path):
    data = path.read_bytes()
    bindings[str(path)] = {"sha256": hashlib.sha256(data).hexdigest(), "size_bytes": len(data)}
    return data


def load(path, name):
    source(path)
    spec = importlib.util.spec_from_file_location(name, path)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module


def functions(path, names, namespace):
    tree = ast.parse(source(path))
    selected = [node for node in tree.body if isinstance(node, ast.FunctionDef) and node.name in names]
    assert {node.name for node in selected} == names
    future = ast.ImportFrom(module="__future__", names=[ast.alias(name="annotations")], level=0)
    tree = ast.fix_missing_locations(ast.Module(body=[future, *selected], type_ignores=[]))
    exec(compile(tree, str(path), "exec"), namespace)

pairs = []
for pair in report["pairs"]:
    for role in ("opening", "candidate", "closing"):
        result = pair[role]
        values = sorted(a["speech_end_to_span_final_ms"] for a in result["anchors"] if a["eligible"] and a["covered"])
        assert result["n"] == len(values) == 6
        assert statistics.median(values) == result["p50_ms"]
        assert values[math.ceil(0.95 * len(values)) - 1] == result["p95_ms"]
        assert result["status"] == "complete"
    assert pair["runtime_identity_errors"] == [[], [], []]
    assert pair["p95_claim_eligible"] is False
    gains = [pair[role]["p50_ms"] - pair["candidate"]["p50_ms"] for role in ("opening", "closing")]
    median_gate = all(
        gain >= 150 or gain >= pair[role]["p50_ms"] * 0.15 for role, gain in zip(("opening", "closing"), gains)
    )
    assert median_gate == ("median_gain_below_gate" not in pair["reasons"])
    assert all(pair["candidate"]["p95_ms"] < pair[role]["p95_ms"] for role in ("opening", "closing"))
    assert pair["status"] == "rejected"
    pairs.append(
        {
            "size": pair["size"],
            "repeat": pair["repeat"],
            "p50_ms": {role: pair[role]["p50_ms"] for role in ("opening", "candidate", "closing")},
            "conservative_median_gain_ms": min(gains),
            "median_gate_passed": median_gate,
            "rejection_reasons": pair["reasons"],
            "eligible_fixed_span_count_per_run": 6,
            "p95_claim_eligible": False,
        }
    )
