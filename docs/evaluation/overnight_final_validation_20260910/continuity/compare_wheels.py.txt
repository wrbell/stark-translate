def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def collect(directory):
    report_path = BASE / directory / 'report.json'
    report_raw = report_path.read_bytes()
    report = json.loads(report_raw)
    artifacts = {row['kind']: row for row in report['artifacts']}
    wheel_evidence = []
    for kind in ('wheel', 'wheel_from_mac', 'wheel_from_sdist'):
        descriptor = artifacts[kind]
        path = Path(descriptor['path'])
        raw = path.read_bytes()
        assert len(raw) == descriptor['size_bytes'], (kind, 'size differs from receipt')
        assert sha(raw) == descriptor['sha256'], (kind, 'hash differs from receipt')
        wheel_evidence.append({'kind': kind, 'path': str(path), 'sha256': sha(raw), 'size_bytes': len(raw)})
    canonical = Path(artifacts['wheel']['path'])
    with zipfile.ZipFile(canonical) as archive:
        infos = [i for i in archive.infolist() if not i.is_dir()]
        assert len(infos) == len({i.filename for i in infos}), 'Duplicate ZIP names'
        assert sum(i.file_size for i in infos) < 32 * 1024 * 1024, 'Unexpected expanded wheel size'
        payloads = {i.filename: archive.read(i.filename) for i in infos}
    actual_hashes = {name: sha(raw) for name, raw in payloads.items()}
    declared = report['runtime_equivalence']['member_sha256']
    for name, expected in declared.items():
        assert actual_hashes[name] == expected, ('Actual member differs from manifest', name)
    runtime_names = {name for name in actual_hashes if '.dist-info/' not in name}
    assert runtime_names == set(declared), ('Manifest does not cover exact non-dist-info member inventory', runtime_names ^ set(declared))
    assert len(declared) == report['runtime_equivalence']['member_count']
    metadata_name = next(name for name in payloads if name.endswith('.dist-info/METADATA'))
    parsed = BytesParser().parsebytes(payloads[metadata_name])
    headers = {key: parsed.get_all(key) for key in parsed.keys()}
    return payloads, {
        'source_revision': report['source_revision'],
        'build_report': {'path': str(report_path), 'sha256': sha(report_raw), 'size_bytes': len(report_raw),
                         'reported_status': report['status'], 'reported_ended_at_utc': report.get('ended_at_utc')},
        'wheels': wheel_evidence, 'runtime_members': sorted(runtime_names),
        'runtime_manifest_matches_actual_members': True,
        'canonical_equals_both_rebuilt_wheels': len({row['sha256'] for row in wheel_evidence}) == 1,
        'wheel_metadata_headers': headers,
    }
